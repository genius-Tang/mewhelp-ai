# Chapter 1 Pure Chat Service Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build and verify a FastAPI/LangChain electric-commerce customer-service backend with stateless multi-turn context, SSE incremental output, managed prompts, bounded message history, and Pydantic-validated after-sales extraction.

**Architecture:** HTTP and SSE concerns live in `app/api`, public contracts in `app/schemas`, configuration/model/prompt/message primitives in `app/core`, and model orchestration in `app/services`. The client resubmits full history; the server validates, trims, adds its system prompt, and streams model chunks without storing sessions.

**Tech Stack:** Python 3.12+, FastAPI 0.139+, LangChain 1.3+, langchain-openai 1.3+, pydantic-settings 2.14+, pytest 9+, HTTPX 0.28+

**Spec:** `docs/superpowers/specs/2026-09-22-ch01-pure-chat-design.md`

## Global Constraints

- Work only inside `MewHelp_AI/code`; do not modify or move `MewHelp_AI/python`.
- Keep the fixed stack: Python, FastAPI, LangChain, and OpenAI-compatible `ChatOpenAI` endpoints.
- Do not add tools, an Agent loop, a database, server-side session storage, or a chat page in this chapter.
- Read concrete library/API behavior from Context7 before implementation; stop for contradictions instead of replacing a fixed choice.
- Never log, commit, or return `CHAT_API_KEY`; commit `.env.example`, never `.env`.
- Treat an upstream non-empty stream chunk as one SSE `token` event; do not re-tokenize or buffer it.
- Use `with_structured_output` with the configured method; do not silently fall back to a different method.
- Append `dev-notes/ch01.md` immediately after plan approval, every completed task, code review, and finish. Each entry records user wording, output/review result, corrections, and failures/rework.
- Use TDD for testable code. For prompt/data work, run the labeled evaluation set instead of forcing prompt text into unit-test assertions.

## Review Focus

- A latest user message larger than the history budget returns HTTP 422 instead of being dropped or truncated; Task 2 and Task 6 pin this behavior.
- A model that yields empty chunks followed by text produces only the non-empty `token` event; Task 4 tests it.
- A stream that fails after one token sends that token, one `error`, and no `done`; Task 6 tests the exact event sequence.
- A structured-output provider/schema mismatch returns sanitized HTTP 502 and never leaks an upstream body or key; Task 5 and Task 6 test it.
- An `.env` with unrelated fields remains loadable while missing/blank required chat settings fail validation without exposing secret values; Task 1 tests it.

```python
from typing import Literal, Self

from pydantic import BaseModel, Field, field_validator, model_validator


class ChatMessage(BaseModel):
    role: Literal["user", "assistant"]
    content: str = Field(min_length=1)

    @field_validator("content")
    @classmethod
    def strip_content(cls, value: str) -> str:
        value = value.strip()
        if not value:
            raise ValueError("content must not be blank")
        return value


class ChatRequest(BaseModel):
    messages: list[ChatMessage] = Field(min_length=1)

    @model_validator(mode="after")
    def require_latest_user_message(self) -> Self:
        if self.messages[-1].role != "user":
            raise ValueError("the last message must have role 'user'")
        return self
```

Create `app/core/messages.py`:

```python
from collections.abc import Callable, Sequence

from langchain_core.messages import AIMessage, BaseMessage, HumanMessage
from langchain_core.messages.utils import count_tokens_approximately, trim_messages

from app.schemas.chat import ChatMessage


TokenCounter = Callable[[Sequence[BaseMessage]], int]


class MessageBudgetExceeded(ValueError):
    """Raised when the latest user message alone exceeds the history budget."""


def to_langchain_messages(messages: list[ChatMessage]) -> list[BaseMessage]:
    converted: list[BaseMessage] = []
    for message in messages:
        if message.role == "user":
            converted.append(HumanMessage(content=message.content))
        else:
            converted.append(AIMessage(content=message.content))
    return converted


def trim_history(
    messages: list[BaseMessage],
    max_tokens: int,
    token_counter: TokenCounter = count_tokens_approximately,
) -> list[BaseMessage]:
    latest = messages[-1]
    if token_counter([latest]) > max_tokens:
        raise MessageBudgetExceeded("latest user message exceeds history token budget")

    trimmed = trim_messages(
        messages,
        strategy="last",
        token_counter=token_counter,
        max_tokens=max_tokens,
        start_on="human",
        end_on="human",
        allow_partial=False,
    )
    return list(trimmed)
```

```python
from langchain_core.messages import BaseMessage
from langchain_core.prompts import ChatPromptTemplate, MessagesPlaceholder


CUSTOMER_SERVICE_SYSTEM_PROMPT = """你是电商平台的智能客服。
请使用用户当前使用的语言，保持简洁、礼貌、明确。
信息不足时先追问，不得编造订单状态、物流进度、退款结果或平台政策。
你当前没有工具和业务系统访问能力，不得声称已经查询数据库或完成退款、退货、换货、补发、维修等操作。
对于超出电商客服范围的问题，简短说明能力边界并引导用户描述电商相关诉求。"""

AFTER_SALES_EXTRACTION_SYSTEM_PROMPT = """你负责从用户的售后描述中提取结构化信息。
只能使用原文明确出现的信息，不得猜测订单号或期望方案。
诉求类型只能选择：退货、退款、换货、补发、维修、物流查询、其他。
原文没有订单号或期望方案时，对应字段返回 null。"""

CHAT_PROMPT = ChatPromptTemplate.from_messages(
    [
        ("system", CUSTOMER_SERVICE_SYSTEM_PROMPT),
        MessagesPlaceholder("history"),
    ]
)

EXTRACTION_PROMPT = ChatPromptTemplate.from_messages(
    [
        ("system", AFTER_SALES_EXTRACTION_SYSTEM_PROMPT),
        ("human", "{description}"),
    ]
)


def build_chat_prompt_messages(history: list[BaseMessage]) -> list[BaseMessage]:
    return CHAT_PROMPT.invoke({"history": history}).to_messages()


def build_extraction_prompt_messages(description: str) -> list[BaseMessage]:
    return EXTRACTION_PROMPT.invoke({"description": description}).to_messages()
```

```python
from collections.abc import AsyncIterator
from typing import Annotated, Any

from fastapi import APIRouter, Depends, HTTPException
from fastapi.responses import StreamingResponse

from app.api.dependencies import get_chat_model
from app.api.sse import encode_sse
from app.core.config import Settings, get_settings
from app.core.messages import MessageBudgetExceeded
from app.schemas.chat import ChatRequest
from app.services.chat import prepare_chat_messages, stream_chat


router = APIRouter(tags=["chat"])


@router.post("/chat/stream")
async def chat_stream(
    request: ChatRequest,
    settings: Annotated[Settings, Depends(get_settings)],
    model: Annotated[Any, Depends(get_chat_model)],
) -> StreamingResponse:
    try:
        messages = prepare_chat_messages(request, settings.history_max_tokens)
    except MessageBudgetExceeded as exc:
        raise HTTPException(status_code=422, detail=str(exc)) from exc

    async def events() -> AsyncIterator[str]:
        try:
            async for token in stream_chat(messages, model):
                yield encode_sse("token", {"content": token})
        except Exception:
            yield encode_sse(
                "error",
                {"code": "upstream_error", "message": "模型服务暂时不可用"},
            )
            return
        yield encode_sse("done", {})

    return StreamingResponse(
        events(),
        media_type="text/event-stream",
        headers={"Cache-Control": "no-cache", "X-Accel-Buffering": "no"},
    )
```

```python
import asyncio
import json
from pathlib import Path

from app.core.config import get_settings
from app.core.model import create_chat_model
from app.schemas.chat import ChatRequest
from app.services.chat import prepare_chat_messages, stream_chat
from app.services.extraction import extract_after_sales


CASES_PATH = Path("evals/ch01_cases.json")
RESULTS_PATH = Path("evals/ch01_results.json")


async def evaluate_case(case, model, settings):
    if case["kind"] == "chat":
        request = ChatRequest(messages=case["messages"])
        messages = prepare_chat_messages(request, settings.history_max_tokens)
        output = "".join([chunk async for chunk in stream_chat(messages, model)])
        passed = (
            any(text in output for text in case["must_contain_any"])
            and all(text not in output for text in case["must_not_contain"])
        )
        return {"id": case["id"], "passed": passed, "actual": output}

    result = await extract_after_sales(
        case["description"],
        model,
        settings.chat_structured_output_method,
    )
    actual = result.model_dump()
    return {
        "id": case["id"],
        "passed": actual == case["expected"],
        "actual": actual,
        "expected": case["expected"],
    }


async def main() -> int:
    settings = get_settings()
    model = create_chat_model(settings)
    cases = json.loads(CASES_PATH.read_text(encoding="utf-8"))
    results = []
    for case in cases:
        try:
            results.append(await evaluate_case(case, model, settings))
        except Exception as exc:
            results.append(
                {"id": case["id"], "passed": False, "error": type(exc).__name__}
            )

    RESULTS_PATH.write_text(
        json.dumps(results, ensure_ascii=False, indent=2),
        encoding="utf-8",
    )
    passed = sum(result["passed"] for result in results)
    print(f"chapter-1 eval: {passed}/{len(results)} passed")
    for result in results:
        print(f"{'PASS' if result['passed'] else 'FAIL'} {result['id']}")
    return 0 if passed == len(results) else 1


if __name__ == "__main__":
    raise SystemExit(asyncio.run(main()))
```
